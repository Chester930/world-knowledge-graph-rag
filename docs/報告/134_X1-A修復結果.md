## 附錄 A：X1 25 格矩陣腳本全文

```python
"""Offline 5-by-5 X1 behavior matrix for report 133.

Each cell starts with a real ``DocumentRecord`` and invokes the public
document-record service operation for the event.  Only folders beneath the
system temporary directory are touched; no worker, server, LLM, or graph
service is started.
"""
from __future__ import annotations

import json
import shutil
import sys
from pathlib import Path


def main(repo: Path, output_name: str) -> None:
    sys.path.insert(0, str(repo))
    from services import document_record_service as records
    from state.document_sm import DocumentEvent, ExtractionStatus

    root = Path(__file__).resolve().parent / "run-data"
    if root.exists():
        shutil.rmtree(root)
    root.mkdir(parents=True)
    matrix = []
    for source_status in ExtractionStatus:
        for event in DocumentEvent:
            folder = root / f"{source_status.value}__{event.value}"
            records.init_record(folder, f"{folder.name}.md", total_chunks=3)
            record = records.read_record(folder)
            assert record is not None
            record.extraction_status = source_status.value
            records._write_record(folder, record)

            if event is DocumentEvent.RESET:
                records.reset_extraction_progress(folder)
            elif event is DocumentEvent.REASSIGN:
                from uuid import UUID
                records.append_assignment(folder, UUID("00000000-0000-0000-0000-000000000001"), "matrix", "manual")
            elif event is DocumentEvent.CHUNK_COMPLETED_PARTIAL:
                records.record_chunk_completed(folder, 1)
            elif event is DocumentEvent.CHUNK_COMPLETED_FULL:
                records.record_chunk_completed(folder, 1)
                records.record_chunk_completed(folder, 2)
                records.record_chunk_completed(folder, 3)
            elif event is DocumentEvent.MARK_FAILED:
                records.mark_extraction_failed(folder)

            observed = records.read_record(folder)
            assert observed is not None
            matrix.append(
                {
                    "source_status": source_status.value,
                    "event": event.value,
                    "result": observed.extraction_status,
                }
            )

    output = {
        "statuses": [status.value for status in ExtractionStatus],
        "events": [event.value for event in DocumentEvent],
        "matrix": matrix,
    }
    destination = Path(__file__).resolve().parent / output_name
    destination.write_text(json.dumps(output, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"output": str(destination), "cells": len(matrix)}, ensure_ascii=False))


if __name__ == "__main__":
    if len(sys.argv) != 3:
        raise SystemExit("usage: python report133_x1_matrix.py REPOSITORY OUTPUT_NAME")
    main(Path(sys.argv[1]).resolve(), sys.argv[2])
```
