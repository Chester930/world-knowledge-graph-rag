from __future__ import annotations
import ast, copy, json, sys
from pathlib import Path

FUNCTION_MAP = {"_strip_type_markers":"strip_type_markers", "_is_contentful_line":"is_contentful_line", "_litm_reorder":"litm_reorder", "_split_fact_lines":"split_fact_lines", "_merge_fact_lines":"merge_fact_lines"}
CONSTANT_NAMES = {"_CONTROLLED_TYPE_TOKEN_PATTERN", "_CONTROLLED_TYPE_LIST_PATTERN", "_TYPE_MARKER_RE"}

class Normalizer(ast.NodeTransformer):
    def visit_FunctionDef(self,node):
        node=self.generic_visit(node); node.name=FUNCTION_MAP.get(node.name,node.name); return node
    def visit_Name(self,node):
        node=self.generic_visit(node); node.id=FUNCTION_MAP.get(node.id,node.id); return node

def funcs(tree): return {n.name:n for n in tree.body if isinstance(n,(ast.FunctionDef,ast.AsyncFunctionDef))}

def assigns(tree):
    out={}
    for n in tree.body:
        if not isinstance(n,(ast.Assign,ast.AnnAssign,ast.AugAssign)): continue
        for t in n.targets if isinstance(n,ast.Assign) else [n.target]:
            if isinstance(t,ast.Name) and t.id in CONSTANT_NAMES: out[t.id]=n
    return out

def dump(node): return ast.dump(Normalizer().visit(copy.deepcopy(node)),annotate_fields=True,include_attributes=False)

def compare(old_funcs,new_funcs,names,old_lookup=None):
    old_lookup=old_lookup or (lambda name:name)
    result={}
    for old_name in names:
        new_name=FUNCTION_MAP[old_name]; old_key=old_lookup(old_name); left=dump(old_funcs[old_key]); right=dump(new_funcs[new_name])
        result[old_name]={"new_name":new_name,"equal":left==right,"old_lines":[old_funcs[old_key].lineno,old_funcs[old_key].end_lineno],"new_lines":[new_funcs[new_name].lineno,new_funcs[new_name].end_lineno]}
        if left != right: raise SystemExit(json.dumps({"kind":"function","name":old_name,"old":left,"new":right},ensure_ascii=False))
    return result

def main():
    agent_tree=ast.parse(Path(sys.argv[1]).read_text(encoding="utf-8")); baseline_fact_tree=ast.parse(Path(sys.argv[2]).read_text(encoding="utf-8")); new_fact_tree=ast.parse(Path(sys.argv[3]).read_text(encoding="utf-8"))
    old_agent=funcs(agent_tree); old_fact=funcs(baseline_fact_tree); new=funcs(new_fact_tree)
    second=compare(old_agent,new,["_split_fact_lines","_merge_fact_lines"]); first=compare(old_fact,new,["_strip_type_markers","_is_contentful_line","_litm_reorder"],lambda name:FUNCTION_MAP[name])
    old_constants=assigns(baseline_fact_tree); new_constants=assigns(new_fact_tree); constants={}
    for name in sorted(CONSTANT_NAMES):
        left=dump(old_constants[name]); right=dump(new_constants[name]); constants[name]={"equal":left==right}
        if left != right: raise SystemExit(json.dumps({"kind":"constant","name":name,"old":left,"new":right},ensure_ascii=False))
    print(json.dumps({"second_cut":second,"first_cut_functions":first,"first_cut_constants":constants,"all_equal":True},ensure_ascii=False,indent=2))

if __name__ == "__main__": main()
