class Normalizer(ast.NodeTransformer):
    def visit_FunctionDef(self, node):
        node = self.generic_visit(node)
        node.name = FUNCTION_MAP.get(node.name, node.name)
        return node

    def visit_Name(self, node):
        node = self.generic_visit(node)
        node.id = FUNCTION_MAP.get(node.id, node.id)
        return node


def functions(tree):
    return {node.name: node for node in tree.body if isinstance(node, ast.FunctionDef)}


def assignments(tree):
    result = {}
    for node in tree.body:
        if not isinstance(node, (ast.Assign, ast.AnnAssign, ast.AugAssign)):
            continue
        targets = node.targets if isinstance(node, ast.Assign) else [node.target]
        for target in targets:
            if isinstance(target, ast.Name) and target.id in CONSTANT_NAMES:
                result[target.id] = node
    return result


def dump(node):
    return ast.dump(
        Normalizer().visit(copy.deepcopy(node)),
        annotate_fields=True,
        include_attributes=False,
    )


def compare_functions(old, new, pairs):
    result = {}
    for old_name, new_name in pairs:
        left = dump(old[old_name])
        right = dump(new[new_name])
        equal = left == right
        result[old_name] = {
            "new_name": new_name,
            "equal": equal,
            "old_lines": [old[old_name].lineno, old[old_name].end_lineno],
            "new_lines": [new[new_name].lineno, new[new_name].end_lineno],
        }
        if not equal:
            raise SystemExit(json.dumps({"name": old_name, "old": left, "new": right}, ensure_ascii=False))
    return result

    if old_value != new_value or serial_json(old_value) != serial_json(new_value):
        raise AssertionError(json.dumps({"label": label, "old": old_value, "new": new_value}, ensure_ascii=False))


def check_case(baseline, telemetry, case, counters):
    triples, facts, prompt_lines, documents, latency_s = case
    check_equal("serialize_none", case, baseline._serialize_document(None), telemetry.serialize_document(None), counters)
    for document in documents.values():
        check_equal("serialize_document", case, baseline._serialize_document(document), telemetry.serialize_document(document), counters)
    check_equal(
        "build_retrieval_telemetry",
        case,
        baseline._build_retrieval_telemetry(triples, facts, latency_s),
        telemetry.build_retrieval_telemetry(triples, facts, latency_s),
        counters,
    )
    check_equal(
        "build_retrieval_trace",
        case,
        baseline._build_retrieval_trace(triples, facts, prompt_lines),
        telemetry.build_retrieval_trace(triples, facts, prompt_lines),
        counters,
    )
    check_equal(
        "serialize_sources",
        case,
        baseline._serialize_sources(triples, facts, "R", documents, {"latency": latency_s}, {"prompt_lines": prompt_lines}),
        telemetry.serialize_sources(triples, facts, "R", documents, {"latency": latency_s}, {"prompt_lines": prompt_lines}),
        counters,
    )


def main():
    baseline = load_baseline(Path(sys.argv[1]).resolve())
    from services.context import telemetry

    rng = random.Random(143)
    random_cases = [random_case(rng, index) for index in range(1000)]
    manual = manual_cases()
    counters = {"serialize_none": 0, "serialize_document": 0, "build_retrieval_telemetry": 0, "build_retrieval_trace": 0, "serialize_sources": 0}
    for case in random_cases + manual:
        check_case(baseline, telemetry, case, counters)
    print(json.dumps({
        "all_equal": True,
        "random_groups": len(random_cases),
        "manual_boundary_groups": len(manual),
        "function_comparisons": counters,
        "json_ensure_ascii_false_compared": True,
    }, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()
