```python
from __future__ import annotations

import ast
import copy
import json
import sys
from pathlib import Path


FUNCTION_MAP = {
    "_strip_type_markers": "strip_type_markers",
    "_is_contentful_line": "is_contentful_line",
    "_litm_reorder": "litm_reorder",
}
CONSTANT_NAMES = {
    "_CONTROLLED_TYPE_TOKEN_PATTERN",
    "_CONTROLLED_TYPE_LIST_PATTERN",
    "_TYPE_MARKER_RE",
}


class Normalizer(ast.NodeTransformer):
    def visit_FunctionDef(self, node: ast.FunctionDef):
        node = self.generic_visit(node)
        node.name = FUNCTION_MAP.get(node.name, node.name)
        return node

    def visit_Name(self, node: ast.Name):
        node = self.generic_visit(node)
        node.id = FUNCTION_MAP.get(node.id, node.id)
        return node


def top_level_functions(tree: ast.Module) -> dict[str, ast.FunctionDef | ast.AsyncFunctionDef]:
    return {
        node.name: node
        for node in tree.body
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef))
    }


def top_level_assignments(tree: ast.Module) -> dict[str, ast.AST]:
    result: dict[str, ast.AST] = {}
    for node in tree.body:
        if not isinstance(node, (ast.Assign, ast.AnnAssign, ast.AugAssign)):
            continue
        targets = node.targets if isinstance(node, ast.Assign) else [node.target]
        for target in targets:
            if isinstance(target, ast.Name) and target.id in CONSTANT_NAMES:
                result[target.id] = node
    return result


def normalized_dump(node: ast.AST) -> str:
    return ast.dump(Normalizer().visit(copy.deepcopy(node)), annotate_fields=True, include_attributes=False)


def main() -> None:
    baseline_path = Path(sys.argv[1]).resolve()
    module_path = Path(sys.argv[2]).resolve()
    baseline_tree = ast.parse(baseline_path.read_text(encoding="utf-8"), filename=str(baseline_path))
    module_tree = ast.parse(module_path.read_text(encoding="utf-8"), filename=str(module_path))
    baseline_functions = top_level_functions(baseline_tree)
    module_functions = top_level_functions(module_tree)
    baseline_assignments = top_level_assignments(baseline_tree)
    module_assignments = top_level_assignments(module_tree)
    function_results = {}
    for old_name, new_name in FUNCTION_MAP.items():
        old_dump = normalized_dump(baseline_functions[old_name])
        new_dump = normalized_dump(module_functions[new_name])
        function_results[old_name] = {
            "new_name": new_name,
            "equal": old_dump == new_dump,
            "baseline_lines": [baseline_functions[old_name].lineno, baseline_functions[old_name].end_lineno],
            "new_lines": [module_functions[new_name].lineno, module_functions[new_name].end_lineno],
        }
        if old_dump != new_dump:
            raise SystemExit(json.dumps({"kind": "function", "name": old_name, "baseline": old_dump, "new": new_dump}, ensure_ascii=False))
    constant_results = {}
    for name in sorted(CONSTANT_NAMES):
        old_dump = normalized_dump(baseline_assignments[name])
        new_dump = normalized_dump(module_assignments[name])
        constant_results[name] = {"equal": old_dump == new_dump}
        if old_dump != new_dump:
            raise SystemExit(json.dumps({"kind": "constant", "name": name, "baseline": old_dump, "new": new_dump}, ensure_ascii=False))
    result = {
        "baseline": str(baseline_path),
        "module": str(module_path),
        "function_results": function_results,
        "constant_results": constant_results,
        "all_equal": True,
    }
    print(json.dumps(result, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()
```

from __future__ import annotations

import importlib.util
import json
import random
import sys
from pathlib import Path


def load_baseline(path: Path):
    spec = importlib.util.spec_from_file_location("report139_baseline_agent", path)
    if spec is None or spec.loader is None:
        raise RuntimeError(f"cannot load {path}")
    module = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = module
    spec.loader.exec_module(module)
    return module


def collect_texts(value, out: list[str]) -> None:
    if isinstance(value, str):
        out.append(value)
    elif isinstance(value, list):
        for item in value:
            collect_texts(item, out)
    elif isinstance(value, dict):
        for item in value.values():
            collect_texts(item, out)


def records(path: Path) -> list[dict]:
    return json.loads(path.read_text(encoding="utf-8"))


def synthetic_strings() -> list[str]:
    rng = random.Random(139)
    atoms = [
        "普通文字",
        "勞工",
        "XORGANIZATION",
        "PERSON",
        "ORGANIZATION",
        "（概念）",
        "（PERSON,PERSON）",
        "（PERSON, PERSON）",
        "（ORGANIZATION，POSITION）",
        "(PERSON,PERSON)",
        "發生",
        "內容",
    ]
    values = []
    for i in range(500):
        left = rng.choice(atoms)
        right = rng.choice(atoms)
        prefix = rng.choice(["", " ", "- ", "  "])
        suffix = rng.choice(["", " ", f" {i}"])
        values.append(f"{prefix}{left}{right}{suffix}")
    return values


def compare_pair(label: str, old, new, old_value, new_value, counters: dict[str, int]) -> None:
    counters[label] = counters.get(label, 0) + 1
    if old_value != new_value:
        raise AssertionError(json.dumps({"label": label, "input": old, "old": old_value, "new": new_value}, ensure_ascii=False))


def main() -> None:
    repo = Path.cwd()
    baseline = load_baseline(Path(sys.argv[1]).resolve())
    from services.context import fact_lines

    all_trace_texts: list[str] = []
    all_prompt_lines: list[list[str]] = []
    for run_name in ("run1", "run2"):
        data = records(repo / "data" / "eval" / "p2_snapshot_20260929" / run_name / "records.json")
        for record in data:
            stage1 = record.get("lineage", {}).get("stage1_retrieval", {})
            stage2 = record.get("lineage", {}).get("stage2_context", {})
            trace = stage1.get("retrieval_trace")
            collect_texts(trace, all_trace_texts)
            lines = stage2.get("prompt_context_lines") or []
            if isinstance(lines, list):
                if all(isinstance(line, str) for line in lines):
                    all_prompt_lines.append(lines)
                else:
                    all_prompt_lines.extend(
                        sequence for sequence in lines
                        if isinstance(sequence, list) and all(isinstance(line, str) for line in sequence)
                    )
    synthetic = synthetic_strings()
    counters: dict[str, int] = {}
    for text in all_trace_texts + [line for lines in all_prompt_lines for line in lines] + synthetic:
        compare_pair("strip_type_markers", text, fact_lines.strip_type_markers,
                     baseline._strip_type_markers(text), fact_lines.strip_type_markers(text), counters)
        subjects = [None, "", "S", baseline._strip_type_markers(text).lstrip("- ").strip()]
        for subject in subjects:
            compare_pair("is_contentful_line", (text, subject), fact_lines.is_contentful_line,
                         baseline._is_contentful_line(text, subject), fact_lines.is_contentful_line(text, subject), counters)
    for lines in all_prompt_lines:
        compare_pair("litm_reorder", lines, fact_lines.litm_reorder,
                     baseline._litm_reorder(lines), fact_lines.litm_reorder(lines), counters)
    for start in range(0, len(synthetic), 10):
        lines = synthetic[start:start + 10]
        compare_pair("litm_reorder", lines, fact_lines.litm_reorder,
                     baseline._litm_reorder(lines), fact_lines.litm_reorder(lines), counters)
    result = {
        "all_equal": True,
        "snapshot_trace_texts": len(all_trace_texts),
        "snapshot_prompt_sequences": len(all_prompt_lines),
        "snapshot_prompt_strings": sum(len(lines) for lines in all_prompt_lines),
        "synthetic_strings": len(synthetic),
        "comparisons": counters,
    }
    print(json.dumps(result, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()
