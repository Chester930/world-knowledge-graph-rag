```python
from __future__ import annotations

import ast
import copy
import json
import sys
from pathlib import Path


FUNCTION_MAP = {
    "_relevant_doc_ids_from_facts": "relevant_doc_ids_from_facts",
    "_intersect_doc_scopes": "intersect_doc_scopes",
    "_resolve_doc_scope": "resolve_doc_scope",
    "_scope_by_source_doc_ids": "scope_by_source_doc_ids",
    "_filter_triples_by_source_doc_ids": "filter_triples_by_source_doc_ids",
    "_filter_facts_by_source_doc_ids": "filter_facts_by_source_doc_ids",
    "_filter_triples_by_relation_type": "filter_triples_by_relation_type",
}


class Normalizer(ast.NodeTransformer):
    def visit_FunctionDef(self, node):
        node = self.generic_visit(node)
        node.name = FUNCTION_MAP.get(node.name, node.name)
        return node

    def visit_Name(self, node):
        node = self.generic_visit(node)
        node.id = FUNCTION_MAP.get(node.id, node.id)
        return node


def functions(path: Path):
    tree = ast.parse(path.read_text(encoding="utf-8"))
    return {node.name: node for node in tree.body if isinstance(node, ast.FunctionDef)}, tree


def normalized_dump(node):
    return ast.dump(
        Normalizer().visit(copy.deepcopy(node)),
        annotate_fields=True,
        include_attributes=False,
    )


def main():
    baseline_dir = Path(sys.argv[1])
    baseline_agent = Path(sys.argv[2])
    scope_path = Path(sys.argv[3])
    baseline_fact = baseline_dir / "baseline_fact_lines.py"
    current_fact = Path(sys.argv[4])
    baseline_telemetry = baseline_dir / "baseline_telemetry.py"
    current_telemetry = Path(sys.argv[5])

    old_functions, _ = functions(baseline_agent)
    new_functions, _ = functions(scope_path)
    pairs = [(old, new) for old, new in FUNCTION_MAP.items()]
    result = {}
    for old_name, new_name in pairs:
        equal = normalized_dump(old_functions[old_name]) == normalized_dump(new_functions[new_name])
        result[old_name] = {"new_name": new_name, "equal": equal}
        if not equal:
            raise SystemExit(json.dumps({"function": old_name, "equal": False}, ensure_ascii=False))

    fact_equal = ast.dump(
        ast.parse(baseline_fact.read_text(encoding="utf-8")),
        include_attributes=False,
    ) == ast.dump(ast.parse(current_fact.read_text(encoding="utf-8")), include_attributes=False)
    telemetry_equal = ast.dump(
        ast.parse(baseline_telemetry.read_text(encoding="utf-8")),
        include_attributes=False,
    ) == ast.dump(
        ast.parse(current_telemetry.read_text(encoding="utf-8")),
        include_attributes=False,
    )
    if not fact_equal or not telemetry_equal:
        raise SystemExit(json.dumps({"fact_lines_equal": fact_equal, "telemetry_equal": telemetry_equal}))

    print(json.dumps({
        "scope_functions": result,
        "fact_lines_module_ast_equal": fact_equal,
        "telemetry_module_ast_equal": telemetry_equal,
        "all_equal": True,
    }, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()
```

def check_case(baseline, case, counters):
    facts, triples, semantic, explicit, seed, allowed, items, rel_type, top_n = case
    old_facts, new_facts = copy.deepcopy(facts), copy.deepcopy(facts)
    compare("relevant_doc_ids_from_facts", baseline._relevant_doc_ids_from_facts, scope.relevant_doc_ids_from_facts, (old_facts,), (new_facts,), counters)
    compare(
        "relevant_doc_ids_from_facts_top_n",
        lambda values: baseline._relevant_doc_ids_from_facts(values, top_n=top_n),
        lambda values: scope.relevant_doc_ids_from_facts(values, top_n=top_n),
        (old_facts,),
        (new_facts,),
        counters,
    )

    old_semantic, new_semantic = set(semantic), set(semantic)
    old_explicit = copy.deepcopy(explicit)
    new_explicit = copy.deepcopy(explicit)
    compare("intersect_doc_scopes", baseline._intersect_doc_scopes, scope.intersect_doc_scopes, (old_semantic, old_explicit), (new_semantic, new_explicit), counters, (old_semantic, new_semantic) if not explicit else None)
    compare("resolve_doc_scope", baseline._resolve_doc_scope, scope.resolve_doc_scope, (set(seed), set(semantic), copy.deepcopy(explicit)), (set(seed), set(semantic), copy.deepcopy(explicit)), counters)

    old_items, new_items = copy.deepcopy(items), copy.deepcopy(items)
    compare("scope_by_source_doc_ids", baseline._scope_by_source_doc_ids, scope.scope_by_source_doc_ids, (old_items, set(allowed), lambda item: item.get("source_doc_id")), (new_items, set(allowed), lambda item: item.get("source_doc_id")), counters, (old_items, new_items))

    old_triples = [triple.model_copy(deep=True) for triple in triples]
    new_triples = [triple.model_copy(deep=True) for triple in triples]
    compare("filter_triples_by_source_doc_ids", baseline._filter_triples_by_source_doc_ids, scope.filter_triples_by_source_doc_ids, (old_triples, set(allowed)), (new_triples, set(allowed)), counters, (old_triples, new_triples))

    old_facts, new_facts = copy.deepcopy(facts), copy.deepcopy(facts)
    compare("filter_facts_by_source_doc_ids", baseline._filter_facts_by_source_doc_ids, scope.filter_facts_by_source_doc_ids, (old_facts, set(allowed)), (new_facts, set(allowed)), counters, (old_facts, new_facts))

    old_triples = [triple.model_copy(deep=True) for triple in triples]
    new_triples = [triple.model_copy(deep=True) for triple in triples]
    compare("filter_triples_by_relation_type", baseline._filter_triples_by_relation_type, scope.filter_triples_by_relation_type, (old_triples, rel_type), (new_triples, rel_type), counters, (old_triples, new_triples) if rel_type is None else None)


def main():
    baseline = load_baseline(Path(sys.argv[1]).resolve())
    rng = random.Random(145)
    random_cases = [random_case(rng, index) for index in range(1000)]
    manual = manual_cases()
    counters = {name: 0 for name in [
        "relevant_doc_ids_from_facts", "relevant_doc_ids_from_facts_top_n", "intersect_doc_scopes",
        "resolve_doc_scope", "scope_by_source_doc_ids", "filter_triples_by_source_doc_ids",
        "filter_facts_by_source_doc_ids", "filter_triples_by_relation_type",
    ]}
    for case in random_cases + manual:
        check_case(baseline, case, counters)

    print(json.dumps({

        "exception_type_and_message_checked": True,
    }, ensure_ascii=False, indent=2))


if __name__ == "__main__":
