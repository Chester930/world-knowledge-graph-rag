## 附錄 A：after SQL 軌跡腳本全文

以下為 `trace_task_queue_sql_after.py` 全文。它是報告124 baseline 腳本的複本，差異已在 §3.1 列明。

```python
"""Record the pre-change SQL trace for report 123 S2.

This is a temporary, offline characterization script.  It imports the existing
task_queue_service only; it does not start Neo4j, Ollama, a server, an import
job, or a re-extraction job.  The SQLite files and JSON output stay beside this
script in the system temporary directory.

Usage:
    python trace_task_queue_sql.py D:/path/to/repository
"""
from __future__ import annotations

import json
import sys
import sqlite3
from pathlib import Path
from unittest.mock import patch


def main(repo: Path) -> None:
    sys.path.insert(0, str(repo))
    from services import task_queue_service as queue
    from state.task_sm import TaskStatus

    root = Path(__file__).resolve().parent
    run_root = root / "run-data"
    run_root.mkdir(parents=True, exist_ok=True)
    traces: list[dict[str, str]] = []
    operations: list[dict[str, object]] = []

    def trace_sql(sql: str) -> None:
        traces.append({"scenario": current_scenario[0], "sql": sql})

    def traced_sqlite_connect(*args: object, **kwargs: object) -> sqlite3.Connection:
        conn = real_sqlite_connect(*args, **kwargs)
        conn.set_trace_callback(trace_sql)
        return conn

    real_sqlite_connect = sqlite3.connect
    real_queue_connect = queue._connect
    current_scenario = ["fixed_sequence"]

    def traced_queue_connect(db_path: Path) -> sqlite3.Connection:
        conn = real_queue_connect(db_path)
        # Keep the _connect wrapper explicit as required by S2.  The sqlite3
        # wrapper above installs the callback early enough to capture schema
        # bootstrap and PRAGMA statements too.
        conn.set_trace_callback(trace_sql)
        return conn

    def call(name: str, function: object, *args: object) -> object:
        result = function(*args)  # type: ignore[operator]
        operations.append({"scenario": current_scenario[0], "operation": name, "result": result})
        return result

    fixed_db = run_root / "fixed_sequence.sqlite3"
    worker_db = run_root / "worker_status_call_forms.sqlite3"
    for db_path in (fixed_db, worker_db):
        if db_path.exists():
            db_path.unlink()
    with patch.object(queue.sqlite3, "connect", side_effect=traced_sqlite_connect):
        with patch.object(queue, "_connect", side_effect=traced_queue_connect):
            call("enqueue([1, 2, 3])", queue.enqueue, fixed_db, "kg-1", "doc.md", [1, 2, 3])
            first = call(
                "claim_next_pending() #1",
                queue.claim_next_pending,
                fixed_db,
                "kg-1",
            )
            second = call(
                "claim_next_pending() #2",
                queue.claim_next_pending,
                fixed_db,
                "kg-1",
            )
            assert first is not None and second is not None
            call(
                "update_status(pending_upload)",
                queue.update_status,
                fixed_db,
                "kg-1",
                "doc.md",
                first[2],
                TaskStatus.PENDING_UPLOAD,
            )
            call(
                "update_status(completed)",
                queue.update_status,
                fixed_db,
                "kg-1",
                "doc.md",
                first[2],
                TaskStatus.COMPLETED,
            )
            call(
                "update_status(failed)",
                queue.update_status,
                fixed_db,
                "kg-1",
                "doc.md",
                second[2],
                TaskStatus.FAILED,
            )
            call("reset_stuck_processing()", queue.reset_stuck_processing, fixed_db)
            call("enqueue([1, 2, 3]) again", queue.enqueue, fixed_db, "kg-1", "doc.md", [1, 2, 3])

            # Record the four status values used by extraction_worker without
            # importing or invoking its Neo4j/LLM-dependent _process_one.
            current_scenario[0] = "worker_status_call_forms"
            call("enqueue([1, 2])", queue.enqueue, worker_db, "kg-1", "doc.md", [1, 2])
            call(
                "worker update_status(processing)",
                queue.update_status,
                worker_db,
                "kg-1",
                "doc.md",
                1,
                TaskStatus.PROCESSING,
            )
            call(
                "worker update_status(failed)",
                queue.update_status,
                worker_db,
                "kg-1",
                "doc.md",
                1,
                TaskStatus.FAILED,
            )
            call(
                "worker update_status(processing)",
                queue.update_status,
                worker_db,
                "kg-1",
                "doc.md",
                2,
                TaskStatus.PROCESSING,
            )
            call(
                "worker update_status(pending_upload)",
                queue.update_status,
                worker_db,
                "kg-1",
                "doc.md",
                2,
                TaskStatus.PENDING_UPLOAD,
            )
            call(
                "worker update_status(completed)",
                queue.update_status,
                worker_db,
                "kg-1",
                "doc.md",
                2,
                TaskStatus.COMPLETED,
            )

    output = {
        "repo": str(repo),
        "script": str(Path(__file__).resolve()),
        "scenarios": ["fixed_sequence", "worker_status_call_forms"],
        "operations": operations,
        "trace": traces,
    }
    output_path = root / "after_sql_trace.json"
    output_path.write_text(json.dumps(output, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"output": str(output_path), "operation_count": len(operations), "trace_count": len(traces)}, ensure_ascii=False))


if __name__ == "__main__":
    if len(sys.argv) != 2:
        raise SystemExit("usage: python trace_task_queue_sql.py REPOSITORY")
    main(Path(sys.argv[1]).resolve())
```
